#!/usr/bin/env bash
# Flint Q8: RTX 4060 dense chain + both cards' expert caches, Vulkan.
set -euo pipefail
FLINT_LOCAL=<COLIBRI_LOCAL>
FLINT_TEST="$FLINT_LOCAL/flint-optimization-20261104"
FLINT_MODEL=<MODEL_DIR>/colibri
FLINT_CLI="$FLINT_LOCAL/../colibri/c/coli"
FLINT_ACTION=${1:-start}
if (($#)); then shift; fi
case "$FLINT_ACTION" in
  stop) exec python3 "$FLINT_CLI" stop --port <LOCAL_PORT> "$@" ;;
  status) exec curl --fail --silent --show-error --max-time 5 <LOCAL_ENDPOINT>/health ;;
  start) ;;
  *) echo "Usage: $0 [start [--no-browser] | stop | status]" >&2; exit 2 ;;
esac
mountpoint -q <MODEL_NVME> || { echo 'The model NVMe is not mounted.' >&2; exit 1; }
[[ -r "$FLINT_MODEL/gguf.json" && -r "$FLINT_MODEL/../flint-Q8_0.gguf" ]] || { echo 'Flint Q8 model files are missing.' >&2; exit 1; }
[[ -x "$FLINT_TEST/vulkan-c/qwen36" && -x "$FLINT_TEST/vk-devices" ]] || { echo 'The verified Vulkan engine or GPU selector is missing.' >&2; exit 1; }
# Prevent inherited experimental settings changing this measured profile.
for FLINT_KEY in ${!COLI_@} ${!QT_@} ${!Q36_@} ${!OMP_@} ${!GOMP_@}; do unset "$FLINT_KEY"; done
unset CUDA_EXPERT_GB CUDA_DENSE HEAT_FILE
export CUDA_DEVICE_ORDER=PCI_BUS_ID
export CUDA_VISIBLE_DEVICES=<GPU_ID>
export COLI_ENGINE="$FLINT_TEST/vulkan-c/qwen36"
export COLI_VK_SHADERS="$FLINT_TEST/vulkan-c/shaders"
export COLI_SETUP_HOME=<COLIBRI_LOCAL>/flint-1080ti-20261104/state-vulkan-expanded
export XDG_CACHE_HOME="$FLINT_LOCAL/cache"
export COLI_USAGE="$COLI_SETUP_HOME/expert-usage"
export COLI_VOCAB_USAGE="$COLI_SETUP_HOME/vocab-usage"
export USAGE_SAVE=0
export COLI_CUDA=0 COLI_VULKAN=1 Q36_MTP=0
export OMP_NUM_THREADS=8 COLI_NO_OMP_TUNE=1 PYTHONUNBUFFERED=1
export COLI_VK_TIER_GB=4.0 COLI_VK_TIER_HOST_GB=40 COLI_VK_RESERVE2_GB=1.5
FLINT_DEVICES=$(python3 - "$FLINT_TEST/vk-devices" <<'PY'
import socket, subprocess, sys
with socket.socket() as sock:
    if sock.connect_ex(('<IP_ADDRESS>',<LOCAL_PORT>)) == 0:
        raise SystemExit('Port <LOCAL_PORT> is in use. Open its Colibri page, or stop that profile first.')
expected=[('5bae6b9c724d008bd7aa3d0bdb86a42f','NVIDIA GeForce RTX 4060',6500),
          ('7eb0c9da3621c00db2a1a95bc16bccba','NVIDIA GeForce GTX 1080 Ti',6000)]
gpus={}
for line in subprocess.check_output(['nvidia-smi','--query-gpu=uuid,name,memory.free','--format=csv,noheader,nounits'],text=True).splitlines():
    uid,name,free=map(str.strip,line.split(','));gpus[uid.removeprefix('GPU-').replace('-','')]=(name,int(free))
for uid,name,free in expected:
    if uid not in gpus or gpus[uid][0]!=name or gpus[uid][1]<free:
        raise SystemExit(f'This profile requires {name} with at least {free} MiB free. No other GPU will be substituted.')
mem=dict(line.split(':',1) for line in open('/proc/meminfo'))
if int(mem['MemAvailable'].split()[0])<50*1024*1024:
    raise SystemExit('This profile requires at least 50 GiB available RAM.')
# Enumerate through the same headless Vulkan API as the engine. vulkaninfo's
# display-oriented order differs on this machine; UUIDs remain unambiguous.
vk={}
for line in subprocess.check_output([sys.argv[1]],text=True).splitlines():
    index,uid,name=line.split('\t',2);vk[uid]=(index,name)
for uid,name,_ in expected:
    if uid not in vk or vk[uid][1]!=name:raise SystemExit(f'Vulkan cannot identify {name}.')
print(vk[expected[0][0]][0],vk[expected[1][0]][0])
PY
)
read -r COLI_VK_DEV COLI_VK_DEV2 <<< "$FLINT_DEVICES"
export COLI_VK_DEV COLI_VK_DEV2
mkdir -p "$COLI_SETUP_HOME"
echo 'Flint Q8 — <LOCAL_ENDPOINT>/ — RTX 4060 + GTX 1080 Ti + system RAM'
echo "Verified Vulkan devices: primary=$COLI_VK_DEV, expert tier=$COLI_VK_DEV2"
exec python3 "$FLINT_CLI" web --model "$FLINT_MODEL" --gpu none --auto-tier \
  --ctx 8192 --ram 48 --cap 256 --no-tune-profile --host <IP_ADDRESS> \
  --port <LOCAL_PORT> --ngen 768 --temp 0 --model-id flint-40b-q8-nvme "$@"
